#version 330 core
layout (location = 0) in vec3 coord;
layout (location = 1) in vec3 normal;
layout (location = 2) in vec3 color;

out vec3 fragment_color;
out vec3 fragment_position;
out vec3 fragment_normal;

uniform mat4 m_rot_psi;
uniform mat4 m_rot_tetta;
uniform mat4 m_rot_gamma;
uniform mat4 m_scale;
uniform mat4 m_trans;

uniform mat4 v_rot_psi;
uniform mat4 v_rot_tetta;
uniform mat4 v_rot_gamma;
uniform mat4 v_scale;
uniform mat4 v_trans;

uniform mat4 projection;

void main() {
	mat4 view = v_scale * v_rot_psi * v_rot_tetta * v_rot_gamma * v_trans;
	mat4 model = m_scale * m_rot_psi * m_rot_tetta * m_rot_gamma * m_trans;
    	gl_Position = projection * view * model * vec4(coord, 1.0);
	fragment_color = color;
	vec4 v1 = view * model * vec4(coord, 1.0);
	vec4 v2 = view * model * vec4(coord + normal, 1.0);
	fragment_position = v1.xyz;
	fragment_normal = normalize(v2.xyz - v1.xyz);
}
